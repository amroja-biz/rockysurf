/*
 * The choice row, etched: the plate's own rule carries selection, and a cut tick replaces the
 * radio dot. A filled radio in a skin where nothing else is filled reads as a foreign part.
 */
function Choice({ selected, onSelect, title, detail, trailing, media }) {
  return (
    <label onClick={onSelect} style={{
      display:'flex', alignItems:'center', gap:'0.75rem', cursor:'pointer', padding:'0.7rem 0.85rem',
      background:'#171420',
      border:'1px solid ' + (selected ? 'rgba(232,195,122,.6)' : 'rgba(252,245,232,.16)'),
      boxShadow: selected ? 'inset 0 0 0 3px rgba(232,195,122,.1)' : 'none',
    }}>
      <svg width="13" height="13" viewBox="0 0 13 13" aria-hidden="true" style={{ flexShrink:0, color: selected ? '#e8c37a' : 'rgba(252,245,232,.3)' }}>
        <rect x="0.5" y="0.5" width="12" height="12" fill="none" stroke="currentColor" strokeWidth="0.9" />
        {selected ? <path d="M2.5 6.8 L5.2 9.6 L10.6 3.2" fill="none" stroke="currentColor" strokeWidth="1.5" /> : null}
      </svg>
      {media}
      <span style={{ flex:1 }}>
        <span style={{ display:'block', color: selected ? '#fcf5e8' : '#cbc5cf', fontSize:'0.875rem', fontWeight:600 }}>{title}</span>
        <span style={{ display:'block', ...QUIET, marginTop:'0.15rem' }}>{detail}</span>
      </span>
      {trailing ? <span style={{ ...MONO, color:'#e8c37a' }}>{trailing}</span> : null}
    </label>
  )
}

function Label({ children, help }) {
  return (
    <div style={{ marginBottom:'0.55rem' }}>
      <div style={{ ...CAPS, color:'#e8c37a', fontWeight:700 }}>{children}</div>
      {help ? <p style={{ fontSize:'0.8125rem', color:'#8a8494', lineHeight:1.5, margin:'0.35rem 0 0', maxWidth:'62ch' }}>{help}</p> : null}
    </div>
  )
}

function CreateScreen({ onCreate }) {
  const [provider, setProvider] = React.useState('hetzner')
  const [size, setSize] = React.useState('medium')
  const [pack, setPack] = React.useState('ai-coding-agents')
  const [name, setName] = React.useState('agent-box')
  const [force, setForce] = React.useState(false)
  const chosen = SIZES.find((s) => s[0] === size)

  return (
    <form onSubmit={(e) => { e.preventDefault(); onCreate() }} style={{ gap:'1.75rem' }}>
      <div>
        <Label help="Rocky Surf holds the credential and calls the API. The resources and the bill are yours.">Provider</Label>
        <div style={{ display:'grid', gap:'0.4rem' }}>
          {PROVIDERS.map(([id, n, d]) => (
            <Choice key={id} selected={provider === id} onSelect={() => setProvider(id)} title={n} detail={d} />
          ))}
        </div>
      </div>

      <div>
        <Label help="Shown here only — nothing on the cloud is renamed.">Name</Label>
        <input value={name} onChange={(e) => setName(e.target.value)} maxLength={63} />
      </div>

      <div>
        <Label help="A t-shirt size resolves to a real machine. Pick an offering directly if you want one.">Size</Label>
        <div style={{ display:'grid', gap:'0.4rem' }}>
          {SIZES.map(([id, n, d, price]) => (
            <Choice key={id} selected={size === id} onSelect={() => setSize(id)} title={n} detail={d} trailing={price} />
          ))}
        </div>
      </div>

      <div>
        <Label help="A list of tools and install scripts. You can read the whole file before trusting it.">Surge Pack</Label>
        <div style={{ display:'grid', gap:'0.4rem' }}>
          {PACKS.map(([id, n, img, d]) => (
            <Choice key={id} selected={pack === id} onSelect={() => setPack(id)} title={n} detail={d}
              media={<PackIcon pack={{ packId:id, name:n, imageUrl:A + img }} />} />
          ))}
        </div>
      </div>

      <div>
        <Label help="One git URL per line, cloned into the box's home directory. Private ones need a token from Settings.">Repositories</Label>
        <div style={{ display:'flex', flexWrap:'wrap', gap:'0.4rem', marginBottom:'0.6rem' }}>
          <Chip>github.com/you/web</Chip><Chip state="added" disabled>github.com/you/api</Chip><Chip state="pending">github.com/you/infra</Chip>
        </div>
        <textarea rows={3} defaultValue={'https://github.com/you/api'} />
        <ul style={{ margin:'0.6rem 0 0', paddingLeft:'1.1rem', fontSize:'0.8125rem', lineHeight:1.55, color:'#8a8494' }}>
          <li style={{ color:'#7ee08f' }}>github.com/you/api — covered by the <code>github.com/you</code> token</li>
          <li style={{ color:'#e8c37a' }}>github.com/you/infra — the config file holds a token this process has not loaded</li>
        </ul>
      </div>

      <CheckboxRow checked={force} onChange={setForce}>Create anyway — I know one of these repositories may not clone.</CheckboxRow>

      {/* The one number the form exists to show, on the one lit plate. */}
      <Plate lit>
        <div style={{ ...CAPS, color:'#e8c37a', fontWeight:700 }}>Resolved offering</div>
        <Swell opacity={0.25} />
        <div style={{ display:'flex', alignItems:'flex-end', justifyContent:'space-between', gap:'1rem', marginTop:'0.7rem' }}>
          <div>
            <div style={{ ...MONO, fontSize:'1rem', color:'#fcf5e8' }}>{chosen[2].split(' · ')[2]} <span className="arch-badge">arm64</span></div>
            <div style={{ ...QUIET, marginTop:'0.35rem' }}>{chosen[2]}</div>
          </div>
          <div style={{ ...MONO, fontSize:'1.5rem', color:'#e8c37a' }}>{chosen[3]}</div>
        </div>
        <p style={{ fontSize:'0.8125rem', color:'#8a8494', margin:'0.7rem 0 0', lineHeight:1.5 }}>
          Estimate from bundled price data, in the currency your provider quotes.
        </p>
      </Plate>

      <div style={{ display:'flex', gap:'0.75rem' }}>
        <button type="submit" className="primary">Create server</button>
        <Button>Preview the install</Button>
      </div>
    </form>
  )
}

function App() {
  const [route, setRoute] = React.useState('create')
  const [status, setStatus] = React.useState('running')
  const title = route === 'create' ? 'Create a server' : 'dev-box'
  return (
    <AppShell title={title} className={route === 'create' ? 'page' : undefined} current={route === 'create' ? '/servers/new' : '/'} markSrc="../../assets/mark-48.png">
      <EtchedDefs />
      {/* exploration control, not part of the design */}
      <div style={{ display:'flex', gap:'0.5rem', marginBottom:'1.25rem' }}>
        <Button onClick={() => setRoute('create')}>Create form</Button>
        <Button onClick={() => { setRoute('detail'); setStatus('running') }}>Server · running</Button>
        <Button onClick={() => { setRoute('detail'); setStatus('provisioning') }}>Server · provisioning</Button>
        <Button onClick={() => { setRoute('detail'); setStatus('failed') }}>Server · failed</Button>
      </div>
      {route === 'create'
        ? <CreateScreen onCreate={() => { setRoute('detail'); setStatus('provisioning') }} />
        : <DetailScreen status={status} onToggle={() => setStatus((s) => (s === 'running' ? 'stopped' : 'running'))} />}
    </AppShell>
  )
}
ReactDOM.createRoot(document.getElementById('root')).render(<App />)
